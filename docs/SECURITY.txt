════════════════════════════════════════════════════════════════════════════════
                           POLICY SQL (for reference)
════════════════════════════════════════════════════════════════════════════════

-- Enable RLS
ALTER TABLE races ENABLE ROW LEVEL SECURITY;
ALTER TABLE judgments ENABLE ROW LEVEL SECURITY;
ALTER TABLE bucket_list ENABLE ROW LEVEL SECURITY;
ALTER TABLE feedback ENABLE ROW LEVEL SECURITY;

-- Read policies
CREATE POLICY "read races" ON races FOR SELECT USING (true);
CREATE POLICY "read judgments" ON judgments FOR SELECT USING (true);
CREATE POLICY "read bucket_list" ON bucket_list FOR SELECT USING (true);
CREATE POLICY "read feedback" ON feedback FOR SELECT USING (true);

-- Write policies
CREATE POLICY "update attending" ON races FOR UPDATE USING (true);
CREATE POLICY "insert feedback" ON feedback FOR INSERT WITH CHECK (true);

════════════════════════════════════════════════════════════════════════════════
                              IF SOMETHING GOES WRONG
════════════════════════════════════════════════════════════════════════════════

Key exposed accidentally?
  → Rotate it immediately in Supabase Dashboard → Settings → API
  → Update .env with new key
  → Update GitHub Secrets with new key

Want to revoke browser access?
  → Delete the RLS policies:
    DROP POLICY "read races" ON races;
  → Or disable the anon key in Supabase settings

Need to see what policies exist?
  → Dashboard → Authentication → Policies
  → Or run: SELECT * FROM pg_policies;
